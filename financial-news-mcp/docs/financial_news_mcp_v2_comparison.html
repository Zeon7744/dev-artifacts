使用示例

1. 采集黄金相关新闻

{
  "method": "tools/call",
  "params": {
    "name": "collect_news",
    "arguments": {
      "category": "commodity",
      "limit": 10,
      "time_range": "24h"
    }
  }
}

2. 分析市场情绪

{
  "method": "tools/call",
  "params": {
    "name": "analyze_sentiment",
    "arguments": {
      "title": "黄金价格创历史新高，突破2100美元",
      "content": "受通胀预期和地缘政治风险影响，黄金价格持续走强"
    }
  }
}

3. 获取投资建议

{
  "method": "tools/call",
  "params": {
    "name": "generate_advice",
    "arguments": {
      "portfolio_value": 100000,
      "risk_tolerance": "moderate",
      "target_return": 12.0
    }
  }
}
